const staging = new RebuildStagingStore({
  get: async (keys) =>
    Object.fromEntries((typeof keys === "string" ? [keys] : keys).map((k) => [k, memory[k]])),
  set: async (rows) => {
    Object.assign(memory, structuredClone(rows));
  },
  remove: async (keys) => {
    for (const k of typeof keys === "string" ? [keys] : keys) delete memory[k];
  }
});

let courseId = "";
let workflowId = "";
let mutations: Promise<unknown>[] = [];
const actions = {
  runtime: { hints: {}, notice: undefined, error: undefined, restoreStage: undefined },
  onMutation: (mutation: unknown) => {
    mutations.push(
      (async () => {
        const c = await store.readCourse(courseId);
        const result = await handler.handle({
          contract: V2_MESSAGE_CONTRACT,
          type: "MUTATE",

        } as never);

      })()
    );
  },

async function paint() {
  const view = await views.build({ surface: "full-page", screen: "IRV-01", courseId });
  renderApp(document.getElementById("root")!, view, actions as never);
  return view;
}
async function mutate(mutation:unknown) {
 const course=(await store.readCourse(courseId))!;
 const result=await handler.handle({contract:V2_MESSAGE_CONTRACT,type:"MUTATE",surface:"full-page",courseId,expectedRevision:course.currentRevision,mutation} as never);
 if(!result.ok) throw Error("ISOLATED_ACTION_FAILED");
}
host.qa052 = {
  async start() {
    await store.initialize();
    const sources = await host.savedSources052();
    courseId = sources[0]!.courseId;
    const stamp = now().toISOString();
    const course: CourseRecord = {
      courseId,
      semesterId: "isolated-052",
      courseCode: "MA6081",
      courseName: "Fundamentals of Project Management - B",
      curriculum: true,
      established: false,
      currentRevision: 0,
      consecutiveCheckFailures: 0,
      updatedAt: stamp
    };
    await store.upsertCourse(course);
    await store.writeDraft({ sources });
    const w = {
      ...createInitialWorkflow(host.qaRunLabel, course, stamp, "initial"),
      phase: "task-b" as const,
      directCoursePolicy: "native-global/1" as const
    };
    workflowId = w.workflowId;
    await store.saveWorkflow(w);
    await engine.run(workflowId);
    const workflow = await store.readWorkflow(workflowId);
    await paint();
    return { state: workflow?.state, phase: workflow?.phase, problem: workflow?.lastErrorDetail };
  },
  async summary() {
    const raw = await store.readSnapshot(courseId),
      trusted = await store.readTrustedSnapshot(courseId);
    return {
      workflow: await store.readWorkflow(workflowId),
      assessments: raw?.assessments.length,
      rules: raw?.constraints.length,
      reviewItems: raw?.reviewItems.length,
      trustedAssessments: trusted?.assessments.length,
      trustedRules: trusted?.constraints.length,
      questions: (await paint()).review?.unresolvedQuestions.length
    };
  },
  async settle() {
    await Promise.all(mutations);
    mutations = [];
    return this.summary();
  },
  async resume() {
    await engine.run(workflowId);
    return this.summary();
  },
  async deepProof() {
    // Only the isolated test DB: exercise actual Human-action handlers before C.
    for (const item of (await store.readSnapshot(courseId))!.reviewItems) {
      await mutate({kind:"ConfirmReviewItem",reviewItemId:item.reviewItemId});
    }
    const initial = (await store.readTrustedSnapshot(courseId))!;
    const exam = initial.assessments.find(a=>a.kind==="exam")!;
    const weight = initial.facts.find(f=>f.assessmentId===exam.assessmentId && f.field==="weight")!;
    if(!weight) throw Error("EXAM_WEIGHT_MISSING");
    await mutate({kind:"EditFact",factId:weight.factId,value:{state:"KNOWN",value:"41%",competingValues:[]}});
    const before = await store.readTrustedSnapshot(courseId);
    const course = (await store.readCourse(courseId))!;
    const check = {...createInitialWorkflow(`${host.qaRunLabel}-check`,course,now().toISOString(),"check"),phase:"task-b" as const,directCoursePolicy:"native-global/1" as const};
    await store.saveWorkflow(check);await engine.run(check.workflowId);
    const after = await store.readTrustedSnapshot(courseId);
    const raw = (await store.readSnapshot(courseId))!;
    const changes=raw.changes;
    const override=(after!.facts.find(f=>f.factId===weight.factId))!.value;
    const rebuild={...createInitialWorkflow(`${host.qaRunLabel}-rebuild`,(await store.readCourse(courseId))!,now().toISOString(),"rebuild"),phase:"task-b" as const,directCoursePolicy:"native-global/1" as const};
    await store.saveWorkflow(rebuild);await engine.run(rebuild.workflowId);
    const staged=await staging.read(courseId,rebuild.workflowId);
    await mutate({kind:"KeepCurrentCourse"});
    return {check:await store.readWorkflow(check.workflowId),before,after,changes,override,rebuild:await store.readWorkflow(rebuild.workflowId),stagingExisted:!!staged,stagingCleared:!(await staging.read(courseId,rebuild.workflowId)),afterKeep:await store.readTrustedSnapshot(courseId)};
  },
  async readPrivate() {
    return {
      snapshot: await store.readSnapshot(courseId),
      staged: await staging.read(courseId, workflowId)
    };
  }
};
